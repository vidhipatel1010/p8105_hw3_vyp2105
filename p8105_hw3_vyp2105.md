The ny_noaa dataset contains daily weather observations from weather
stations across New York State. It includes station identifiers,
observation dates, precipitation, snowfall, snow depth, maximum
temperature, and minimum temperature. The dataset contains 2,595,176
observations and 7 variables. Missing data may be an issue because some
weather measurements are not recorded for every station on every day.

``` r
# Separate the date into year, month, and day
noaa_clean =
  ny_noaa |>
  separate(date, into = c("year", "month", "day"), sep = "-") |>
  mutate(
    # Convert date components to integers
    year = as.integer(year),
    month = as.integer(month),
    day = as.integer(day),

    # Convert precipitation from tenths of mm to mm
    prcp = prcp / 10,

    # Convert temperatures from tenths of degrees Celsius
    tmax = as.numeric(tmax) / 10,
    tmin = as.numeric(tmin) / 10,

    # Keep snowfall and snow depth in their original units
    snow = as.numeric(snow),
    snwd = as.numeric(snwd)
  )
```

## Average maximum temperature in January and July

``` r
# Calculate average maximum temperature by station, year, and month
noaa_monthly = noaa_clean |>
  filter(month %in% c(1, 7)) |>
  group_by(id, year, month) |>
  summarize(
    mean_tmax = mean(tmax, na.rm = TRUE),
    .groups = "drop"
  ) |>
  filter(!is.nan(mean_tmax))

# Plot January and July temperatures across years
ggplot(noaa_monthly, aes(x = year, y = mean_tmax, group = id)) +
  geom_line(alpha = 0.08) +
  facet_wrap(~month, ncol = 2,
             labeller = labeller(month = c("1" = "January",
                                           "7" = "July"))) +
  labs(
    title = "Average Maximum Temperature by Station",
    x = "Year",
    y = "Mean Maximum Temperature (°C)"
  ) +
  theme_minimal()
```

## Temperature relationship and snowfall distributions

``` r
# Plot 1: Relationship between maximum and minimum temperature
temp_plot <- noaa_clean |>
  filter(!is.na(tmax), !is.na(tmin)) |>
  ggplot(aes(x = tmin, y = tmax)) +
  geom_bin2d() +
  labs(
    title = "Maximum vs Minimum Temperature",
    x = "Minimum Temperature (°C)",
    y = "Maximum Temperature (°C)"
  ) +
  theme_minimal()


# Plot 2: Snowfall distribution by year
snow_plot <- noaa_clean |>
  filter(!is.na(snow), snow > 0, snow < 100) |>
  ggplot(aes(x = factor(year), y = snow)) +
  geom_boxplot(outlier.alpha = 0.2) +
  labs(
    title = "Snowfall Distribution by Year",
    x = "Year",
    y = "Snowfall (mm)"
  ) +
  scale_x_discrete(breaks = seq(1980, 2010, by = 5)) +
  theme_minimal() +
  theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 8))


# Display both plots side by side
```

year. Zero snowfall and values of 100 mm or greater are excluded to
focus on smaller snowfall amounts. The boxplots allow comparison of
yearly distributions, but the large number of years makes individual
